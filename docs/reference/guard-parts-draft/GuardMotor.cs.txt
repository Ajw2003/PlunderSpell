using System.Collections.Generic;
using Plunderspell.Status;
using UnityEngine;
using UnityEngine.AI;

namespace Plunderspell.Guards
{
    /// <summary>
    /// How a guard gets about: the dynamic body the NavMeshAgent steers, the stuck watchdog, NavMesh
    /// snapping, the patrol / sweep / look-around / wander helpers, the chase, Levo lift and fall, and
    /// shoves. A plain class owned by <see cref="CastleGuard"/>, which forwards the Unity callbacks
    /// (FixedUpdate, Update) to it. The Rigidbody is found once in the constructor: nothing here asks
    /// Unity for a component per frame (#201).
    ///
    /// The reasoning for "never standing still" is in docs/4-systems/raid.md, "Guards that keep moving".
    /// </summary>
    public class GuardMotor
    {
        private const float k_bodyMass = 80f;
        private const float k_shoveSeconds = 0.25f;
        private const float k_liftHeight = 1.8f;
        private const float k_liftSpeed = 3f;
        private const float k_fallDamagePerMetre = 9f;

        /// <summary>How far to look for the nearest NavMesh point when snapping a destination.</summary>
        private const float k_snapRadius = 6f;

        /// <summary>A destination that moves less than this is not sent to the agent again:
        /// SetDestination restarts path computation.</summary>
        private const float k_resendDistance = 0.75f;

        private const float k_strikeStopFraction = 0.8f;

        private static PhysicsMaterial s_slippery;

        private readonly CastleGuard _owner;
        private readonly NavMeshAgent _agent;
        private readonly Transform _transform;
        private readonly Rigidbody _body;

        private bool _bodyDriven;
        private Vector3 _shoveVelocity;
        private float _shoveUntil;

        private bool _floating;
        private bool _falling;
        private float _floatBaseY;
        private float _fallFromY;
        private float _fallStartedAt;
        private float _fallSpeed;

        private float _tickDelta;
        private bool _seenThisTick;
        private float _moveSpeed;
        private Vector3? _steerTarget;

        private NavMeshPath _scratchPath;
        private Vector3 _sentDestination;
        private bool _hasSentDestination;
        private Vector3 _watchGoal;
        private Vector3 _watchAnchor;
        private float _stuckTime;
        private int _stuckStage;
        private Vector3? _detour;
        private bool _gaveUp;

        private Vector3 _sweepCentre;
        private int _sweepIndex;
        private Vector3? _sweepGoal;
        private Vector3? _wanderGoal;
        private Vector3? _home;
        private int _patrolIndex;

        /// <summary>Where the guard is heading, when it has somewhere to be.</summary>
        public Vector3? Destination { get; private set; }

        /// <summary>Where the guard is going to investigate, if anywhere.</summary>
        public Vector3? InvestigationTarget { get; set; }

        /// <summary>Seconds left of turning on the spot at an investigation target.</summary>
        public float LookTimeLeft { get; set; }

        /// <summary>True while Levo holds this guard up or it is still falling back down.</summary>
        public bool IsAirborne => _floating || _falling;

        /// <summary>
        /// Builds the motor for <paramref name="owner"/>. The guard is a real physics participant
        /// (#200): a finite-mass dynamic body the agent only plans for. It starts kinematic; the
        /// server flips it dynamic in <see cref="FixedStep"/>, a client leaves it kinematic.
        /// </summary>
        public GuardMotor(CastleGuard owner, NavMeshAgent agent)
        {
            _owner = owner;
            _agent = agent;
            _transform = owner.transform;
            owner.TryGetComponent(out _body);

            if (_agent == null)
                return;
            if (_body == null)
                _body = owner.gameObject.AddComponent<Rigidbody>();
            _body.mass = k_bodyMass;
            _body.freezeRotation = true;
            _body.useGravity = true;
            _body.interpolation = RigidbodyInterpolation.Interpolate;
            _body.isKinematic = true;
            foreach (Collider c in owner.GetComponentsInChildren<Collider>())
                if (!c.isTrigger)
                    c.sharedMaterial = Slippery;
        }

        /// <summary>No friction against walls and floor: the body is steered by velocity, so friction
        /// only ever holds it against a wall it brushes (#200).</summary>
        private static PhysicsMaterial Slippery
        {
            get
            {
                if (s_slippery == null)
                    s_slippery = new PhysicsMaterial("GuardSlippery")
                    {
                        dynamicFriction = 0f,
                        staticFriction = 0f,
                        bounciness = 0f,
                        frictionCombine = PhysicsMaterialCombine.Minimum,
                        bounceCombine = PhysicsMaterialCombine.Minimum,
                        hideFlags = HideFlags.HideAndDontSave
                    };
                return s_slippery;
            }
        }

        private bool OnMesh => _agent != null && _agent.isOnNavMesh;

        // -----------------------------------------------------------------------------------------
        // The body
        // -----------------------------------------------------------------------------------------

        /// <summary>
        /// Guards pass through each other (as their kinematic bodies always did) so two dynamic bodies
        /// cannot jam, while still colliding with players and walls (#200).
        /// </summary>
        public void IgnoreOtherGuards(IReadOnlyList<Component> guards)
        {
            Collider[] mine = _owner.GetComponentsInChildren<Collider>();
            for (int g = 0; g < guards.Count; g++)
            {
                if (!(guards[g] is CastleGuard other) || other == _owner)
                    continue;
                foreach (Collider theirs in other.GetComponentsInChildren<Collider>())
                    foreach (Collider c in mine)
                        Physics.IgnoreCollision(c, theirs, true);
            }
        }

        /// <summary>
        /// Server only, from FixedUpdate. The agent plans (updatePosition off, so it never writes the
        /// transform, which is what made agent and physics fight in #104) and the body follows the
        /// agent's desired velocity. Because the body is dynamic with finite mass, a contact with a
        /// player or a wall stops it instead of crushing a player into the wall (#200).
        /// </summary>
        public void FixedStep()
        {
            if (!CanDriveBody())
            {
                _bodyDriven = false;
                return;
            }

            if (!_bodyDriven)
                TakeBody();

            Vector3 want = _agent.isStopped ? Vector3.zero : _agent.desiredVelocity;
            want.y = 0f;
            if (Time.time < _shoveUntil)
                want += _shoveVelocity;
            Vector3 v = _body.linearVelocity;
            _body.linearVelocity = new Vector3(want.x, v.y, want.z);
            _agent.nextPosition = _body.position;
        }

        private bool CanDriveBody() =>
            _body != null && _agent != null && !IsAirborne && _agent.enabled && _agent.isOnNavMesh && !_owner.IsDead;

        private void TakeBody()
        {
            _bodyDriven = true;
            _agent.updatePosition = false;
            _body.isKinematic = false;
            _body.linearVelocity = Vector3.zero;
            _agent.nextPosition = _body.position;
        }

        /// <summary>Knocks the guard back by <paramref name="metres"/> over a moment; walls and bodies stop it.</summary>
        public void Shove(Vector3 metres)
        {
            _shoveVelocity = new Vector3(metres.x, 0f, metres.z) / k_shoveSeconds;
            _shoveUntil = Time.time + k_shoveSeconds;
        }

        /// <summary>A client's guard is moved by the replicated transform, so its own agent is switched off.</summary>
        public void DisableAgent()
        {
            if (_agent != null)
                _agent.enabled = false;
        }

        // -----------------------------------------------------------------------------------------
        // Levitation (Levo)
        // -----------------------------------------------------------------------------------------

        /// <summary>
        /// Lifts the guard while it levitates and drops it when the spell ends. The body is kinematic
        /// (the NavMeshAgent owns the position), so an impulse cannot lift it: the agent is suspended
        /// and the transform raised directly, then the body falls under real gravity and takes damage
        /// for the height. Returns true while the AI must not run.
        /// </summary>
        public bool UpdateLevitation(float deltaTime)
        {
            StatusEffectReceiver status = _owner.Status;
            if (status != null && status.IsLevitating)
            {
                Lift(deltaTime);
                return true;
            }
            if (_floating)
            {
                BeginFall();
                return true;
            }
            return _falling && Fall(deltaTime);
        }

        private void Lift(float deltaTime)
        {
            if (!_floating)
                BeginLift();

            Vector3 p = _transform.position;
            p.y = Mathf.MoveTowards(p.y, _floatBaseY + k_liftHeight, k_liftSpeed * deltaTime);
            _transform.position = p;
            _transform.Rotate(Vector3.up, 45f * deltaTime, Space.World);
        }

        private void BeginLift()
        {
            _floating = true;
            _falling = false;
            _floatBaseY = _transform.position.y;
            if (_agent != null)
            {
                _agent.enabled = false;
                _agent.updatePosition = true;
            }
            _bodyDriven = false;
            if (_body != null)
            {
                _body.isKinematic = true;
                _body.freezeRotation = true;
            }
        }

        private void BeginFall()
        {
            _floating = false;
            _falling = true;
            _fallFromY = _transform.position.y;
            _fallStartedAt = Time.time;
            _fallSpeed = 0f;
            if (_body == null)
                return;
            _body.isKinematic = false;
            _body.useGravity = true;
            _body.freezeRotation = true;
            _body.linearVelocity = Vector3.zero;
        }

        /// <summary>Returns true while the guard is still in the air.</summary>
        private bool Fall(float deltaTime) => _body == null ? FallByHand(deltaTime) : FallWithBody();

        /// <summary>
        /// Raid guards have no Rigidbody, so fall by hand, straight back down to where the lift
        /// started: the guard was raised vertically with its agent off, so that is the floor.
        /// Landing at once here cost the drop nothing (#106).
        /// </summary>
        private bool FallByHand(float deltaTime)
        {
            _fallSpeed -= Physics.gravity.y * deltaTime;
            Vector3 p = _transform.position;
            p.y = Mathf.Max(_floatBaseY, p.y - _fallSpeed * deltaTime);
            _transform.position = p;
            if (p.y > _floatBaseY)
                return true;
            Land();
            return false;
        }

        private bool FallWithBody()
        {
            bool settled = Time.time - _fallStartedAt > 0.25f && Mathf.Abs(_body.linearVelocity.y) < 0.05f;
            bool timedOut = Time.time - _fallStartedAt > 3f;
            if (!settled && !timedOut)
                return true;
            Land();
            return false;
        }

        private void Land()
        {
            _falling = false;
            if (_body != null)
                _body.isKinematic = true;

            HurtByFall();

            _bodyDriven = false;
            if (_agent != null && !_owner.IsDead)
                ReturnToMesh();
        }

        private void HurtByFall()
        {
            float height = Mathf.Max(0f, _fallFromY - _transform.position.y);
            if (height <= 0.5f)
                return;
            StatusEffectReceiver status = _owner.Status;
            GameObject blame = status != null ? status.LevitatedBy : null;
            Damage.Apply(_owner, height * k_fallDamagePerMetre, _owner.gameObject, blame,
                _transform.position + Vector3.up * 0.2f, DamageKind.Impact);
        }

        private void ReturnToMesh()
        {
            _agent.updatePosition = true;
            _agent.enabled = true;
            if (NavMesh.SamplePosition(_transform.position, out NavMeshHit hit, 2f, NavMesh.AllAreas))
                _agent.Warp(hit.position);
        }

        // -----------------------------------------------------------------------------------------
        // Per-tick setup
        // -----------------------------------------------------------------------------------------

        /// <summary>Tells the motor how long this tick is and whether the guard can see its target.</summary>
        public void BeginTick(float deltaTime, bool seesTarget)
        {
            _tickDelta = deltaTime;
            _seenThisTick = seesTarget;
            _home ??= _transform.position;
        }

        /// <summary>Sets the agent's speed for this tick.</summary>
        public void SetSpeed(float speed)
        {
            _moveSpeed = speed;
            if (_agent != null && _agent.isOnNavMesh)
                _agent.speed = speed;
        }

        // -----------------------------------------------------------------------------------------
        // What to do: chase, search, investigate, patrol, wander
        // -----------------------------------------------------------------------------------------

        /// <summary>Stands still: no destination, agent stopped.</summary>
        public void Stop() => MoveTo(null);

        /// <summary>
        /// Closes on <paramref name="target"/> and halts a fraction of <paramref name="reach"/> short
        /// of it, so the guard can strike but never walks its body into the player: the depenetration
        /// was shoving players through walls (#200). The agent's own stoppingDistance did not hold it
        /// (it braked into the target regardless), so the destination is the stopping point.
        /// </summary>
        public void ChaseToward(Vector3 target, float reach, float deltaTime)
        {
            float stopShort = reach * k_strikeStopFraction;

            Vector3 away = _transform.position - target;
            away.y = 0f;
            if (away.magnitude <= stopShort)
            {
                // In position: stand and face the target instead of pressing on.
                MoveTo(null);
                FaceToward(-away, deltaTime);
                return;
            }

            MoveTo(target + away.normalized * stopShort);
        }

        private void FaceToward(Vector3 toward, float deltaTime)
        {
            if (toward.sqrMagnitude <= 0.0001f || deltaTime <= 0f)
                return;
            Quaternion facing = Quaternion.LookRotation(toward, Vector3.up);
            _transform.rotation = Quaternion.RotateTowards(_transform.rotation, facing, _owner.TurnSpeed * deltaTime);
        }

        /// <summary>Makes <paramref name="centre"/> the middle of the next sweep and restarts it.</summary>
        public void ResetSweep(Vector3 centre)
        {
            _sweepCentre = centre;
            _sweepIndex = 0;
            _sweepGoal = null;
        }

        /// <summary>Goes to the sweep centre, then sweeps reachable points around it instead of
        /// parking on it (#193).</summary>
        public void Search()
        {
            if (!_sweepGoal.HasValue)
                _sweepGoal = NextSweepGoal();

            MoveTo(_sweepGoal.Value);
            if (HasArrivedAt(_sweepGoal.Value) || TakeGaveUp())
                _sweepGoal = null;
        }

        private Vector3 NextSweepGoal()
        {
            if (_sweepIndex == 0)
            {
                _sweepIndex = 1;
                return _sweepCentre;
            }

            // A few tries: a ring point inside a wall or cut off from here is skipped, not walked at.
            for (int attempt = 0; attempt < 4; attempt++)
            {
                Vector3? point = Reachable(_sweepCentre + GuardBrain.SweepOffset(_sweepIndex++));
                if (point.HasValue)
                    return point.Value;
            }
            return _sweepCentre;
        }

        /// <summary>Walks to the noise, looks around for a moment, then the state falls back to the
        /// route. Turning on the spot rather than standing there is the point (#193).</summary>
        public void Investigate(float deltaTime)
        {
            if (!InvestigationTarget.HasValue)
                return;

            if (LookTimeLeft > 0f)
            {
                LookAround(deltaTime);
                return;
            }

            Vector3 target = InvestigationTarget.Value;
            MoveTo(target);
            if (HasArrivedAt(target))
                LookTimeLeft = GuardBrain.LookAroundSeconds;
            else if (TakeGaveUp())
                InvestigationTarget = null;
        }

        private void LookAround(float deltaTime)
        {
            MoveTo(null);
            _transform.Rotate(Vector3.up, _owner.TurnSpeed * 0.5f * deltaTime, Space.World);
            LookTimeLeft -= deltaTime;
            if (LookTimeLeft <= 0f)
                InvestigationTarget = null;  // nothing here; back to the route
        }

        /// <summary>Walks the patrol route in order, or wanders near the post when there is no route.</summary>
        public void Patrol()
        {
            if (UsableWaypoints() < 2)
            {
                Wander();
                return;
            }

            IReadOnlyList<Transform> route = _owner.PatrolRoute;
            Transform point = route[_patrolIndex % route.Count];
            if (point == null)
            {
                _patrolIndex++;
                return;
            }

            MoveTo(point.position);
            // An unreachable point is skipped rather than walked at forever.
            if (HasArrivedAt(point.position) || TakeGaveUp())
                _patrolIndex = (_patrolIndex + 1) % route.Count;
        }

        private int UsableWaypoints()
        {
            int usable = 0;
            IReadOnlyList<Transform> route = _owner.PatrolRoute;
            for (int i = 0; i < route.Count; i++)
                if (route[i] != null)
                    usable++;
            return usable;
        }

        /// <summary>A guard with no route to walk drifts between reachable points near where it was
        /// posted, instead of standing its post.</summary>
        private void Wander()
        {
            if (!_wanderGoal.HasValue)
                _wanderGoal = PickWanderGoal();

            MoveTo(_wanderGoal.Value);
            if (HasArrivedAt(_wanderGoal.Value) || TakeGaveUp())
                _wanderGoal = null;
        }

        private Vector3 PickWanderGoal()
        {
            Vector2 around = Random.insideUnitCircle.normalized * Random.Range(3f, 8f);
            Vector3 centre = _home ?? _transform.position;
            return Reachable(centre + new Vector3(around.x, 0f, around.y)) ?? centre;
        }

        // -----------------------------------------------------------------------------------------
        // Moving, snapping and the stuck watchdog (#193)
        // -----------------------------------------------------------------------------------------

        /// <summary>The nearest NavMesh point to <paramref name="point"/>, or the point itself where
        /// there is no mesh to snap to (a player on a table, a noise inside a wall).</summary>
        public Vector3 SnapToMesh(Vector3 point)
        {
            if (OnMesh && NavMesh.SamplePosition(point, out NavMeshHit hit, k_snapRadius, NavMesh.AllAreas))
                return hit.position;
            return point;
        }

        /// <summary>
        /// A point this guard can actually walk to: snapped to the mesh and path-checked when there is
        /// one, the point unchanged when there is not, null when it is cut off.
        /// </summary>
        public Vector3? Reachable(Vector3 point)
        {
            if (!OnMesh)
                return point;
            if (!NavMesh.SamplePosition(point, out NavMeshHit hit, 2.5f, NavMesh.AllAreas))
                return null;
            _scratchPath ??= new NavMeshPath();
            if (!NavMesh.CalculatePath(_transform.position, hit.position, NavMesh.AllAreas, _scratchPath)
                || _scratchPath.status != NavMeshPathStatus.PathComplete)
                return null;
            return hit.position;
        }

        private bool HasArrivedAt(Vector3 position) =>
            Vector3.Distance(_transform.position, SnapToMesh(position)) <= _owner.ArrivalDistance;

        /// <summary>True once, when the watchdog gave up on the destination it was watching.</summary>
        private bool TakeGaveUp()
        {
            bool gaveUp = _gaveUp;
            _gaveUp = false;
            return gaveUp;
        }

        private void MoveTo(Vector3? destination)
        {
            Destination = destination;

            if (!destination.HasValue)
            {
                StopMoving();
                return;
            }

            Vector3 goal = destination.Value;
            WatchProgress(goal);
            Vector3 heading = SnapToMesh(_detour ?? goal);

            if (OnMesh)
                SendToAgent(heading);
            else
                _steerTarget = heading;
        }

        private void StopMoving()
        {
            _hasSentDestination = false;
            _stuckTime = 0f;
            _detour = null;
            if (OnMesh)
                _agent.isStopped = true;
            _steerTarget = null;
        }

        private void SendToAgent(Vector3 heading)
        {
            _agent.isStopped = false;
            bool moved = (heading - _sentDestination).sqrMagnitude > k_resendDistance * k_resendDistance;
            bool pathLost = !_agent.hasPath && !_agent.pathPending;
            if (_hasSentDestination && !moved && !pathLost)
                return;
            _agent.SetDestination(heading);
            _sentDestination = heading;
            _hasSentDestination = true;
        }

        /// <summary>
        /// Notices a guard that has a destination and is going nowhere: under
        /// <see cref="GuardBrain.StuckProgress"/> metres in <see cref="GuardBrain.StuckSeconds"/>.
        /// The first time it forces a fresh path; the second time it heads for a reachable point near
        /// the goal; and if there is none it gives the goal up, so the caller moves on to the next one.
        /// </summary>
        private void WatchProgress(Vector3 goal)
        {
            if (_moveSpeed <= 0.01f || _tickDelta <= 0f)
                return;   // a turret is meant not to move

            if (_seenThisTick)
            {
                // Closing on a target it can see, and standing to strike it, is not being stuck.
                ClearStuck();
                _detour = null;
                return;
            }

            RetargetWatch(goal);
            if (ReachedWatchTarget(goal) || MadeProgress())
                return;

            _stuckTime += _tickDelta;
            if (_stuckTime >= GuardBrain.StuckSeconds)
                EscalateStuck(goal);
        }

        private void ClearStuck()
        {
            _stuckTime = 0f;
            _stuckStage = 0;
        }

        /// <summary>A goal that has moved on is a new watch: the old stuck clock means nothing for it.</summary>
        private void RetargetWatch(Vector3 goal)
        {
            if ((goal - _watchGoal).sqrMagnitude <= 1.5f * 1.5f)
                return;
            _watchGoal = goal;
            _watchAnchor = _transform.position;
            ClearStuck();
            _detour = null;
        }

        private bool ReachedWatchTarget(Vector3 goal)
        {
            if (!HasArrivedAt(_detour ?? goal))
                return false;
            _watchAnchor = _transform.position;
            _stuckTime = 0f;
            if (_detour.HasValue)
            {
                _detour = null;
                _hasSentDestination = false;
            }
            return true;
        }

        private bool MadeProgress()
        {
            Vector3 here = _transform.position;
            Vector3 moved = here - _watchAnchor;
            moved.y = 0f;
            if (GuardBrain.IsStuck(moved.magnitude))
                return false;
            _watchAnchor = here;
            ClearStuck();
            return true;
        }

        private void EscalateStuck(Vector3 goal)
        {
            _stuckTime = 0f;
            _watchAnchor = _transform.position;
            _hasSentDestination = false;   // resend: a fresh path
            if (++_stuckStage == 1)
                return;

            _stuckStage = 0;
            Vector2 around = Random.insideUnitCircle.normalized * Random.Range(2f, 5f);
            Vector3? detour = OnMesh ? Reachable(goal + new Vector3(around.x, 0f, around.y)) : null;
            if (detour.HasValue)
                _detour = detour;
            else
                _gaveUp = true;
        }

        /// <summary>
        /// Straight-line movement for a guard without a NavMeshAgent on a baked mesh.
        ///
        /// A procedurally generated castle has no baked NavMesh until the seed is known, so without
        /// this every guard in a generated raid can see and hear but never take a step. It walks into
        /// walls where an agent would route around them; that is the honest trade for guards that
        /// move at all. Bake a mesh at runtime and they use it instead, automatically.
        /// </summary>
        public void Steer(float deltaTime, float speed)
        {
            if (!_steerTarget.HasValue || speed <= 0f || deltaTime <= 0f)
                return;

            Vector3 target = _steerTarget.Value;
            Vector3 flat = new Vector3(target.x - _transform.position.x, 0f, target.z - _transform.position.z);
            float arrival = _owner.ArrivalDistance;
            if (flat.sqrMagnitude <= arrival * arrival)
                return;

            Vector3 direction = flat.normalized;
            _transform.position += direction * (speed * deltaTime);

            Quaternion facing = Quaternion.LookRotation(direction, Vector3.up);
            _transform.rotation = Quaternion.RotateTowards(_transform.rotation, facing, _owner.TurnSpeed * deltaTime);
        }
    }
}
