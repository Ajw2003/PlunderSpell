using System.Collections.Generic;
using Plunderspell.Alarm;
using UnityEngine;

namespace Plunderspell.Guards
{
    /// <summary>
    /// What a guard can see: the cone, the range by alarm level, the arrival grace and the line of
    /// sight. A plain class owned by <see cref="CastleGuard"/> (not a component) so the prefab, the
    /// PurrNet spawn and every test that adds a CastleGuard keep working unchanged.
    ///
    /// The look runs about <see cref="LooksPerSecond"/> times a second, staggered per guard, and the
    /// result is cached between looks: it was one linecast per guard per intruder every frame (#201).
    /// </summary>
    public class GuardSight
    {
        /// <summary>How long after a raid starts a calm garrison cannot see the players.</summary>
        public const float ArrivalGraceSeconds = 20f;

        /// <summary>How many times a second a guard actually looks.</summary>
        public const float LooksPerSecond = 12f;

        private static float s_arrivalGraceEndsAt;

        private readonly CastleGuard _owner;
        private readonly float _phase;
        private float _untilNextLook;
        private bool _hasLooked;
        private Transform _cached;

        /// <summary>Builds the sight for <paramref name="owner"/>. The phase spreads guards across the interval.</summary>
        public GuardSight(CastleGuard owner)
        {
            _owner = owner;
            // Per-guard offset so the whole garrison never looks on the same frame.
            _phase = Mathf.Abs(owner.GetInstanceID() % 1000) / 1000f / LooksPerSecond;
        }

        /// <summary>
        /// Starts the arrival grace: until it runs out, and while the alarm is still calm, no guard
        /// sees a player. Guards still hear, so noise still draws them (seen in co-op testing,
        /// 2026-09-23, a patrol killed players still reading the HUD).
        /// </summary>
        public static void BeginArrivalGrace() => s_arrivalGraceEndsAt = Time.time + ArrivalGraceSeconds;

        /// <summary>Ends the arrival grace now. For tests: the grace is process-wide.</summary>
        public static void EndArrivalGrace() => s_arrivalGraceEndsAt = 0f;

        /// <summary>Test seam: the next <see cref="Update"/> looks at once instead of waiting for its turn.</summary>
        public void ForceLook() => _untilNextLook = 0f;

        /// <summary>
        /// The nearest intruder seen as of the last look, or null. Looks again only when the guard's
        /// turn has come; between looks it returns the cached answer.
        /// </summary>
        public Transform Update(AlarmState alarm, float deltaTime)
        {
            if (!CanLook(alarm))
            {
                _cached = null;
                return null;
            }

            _untilNextLook -= deltaTime;
            if (_hasLooked && _untilNextLook > 0f)
                return _cached != null ? _cached : null;

            // The first gap is the guard's phase, so guards that spawn together drift apart.
            _untilNextLook = _hasLooked ? 1f / LooksPerSecond : _phase;
            _hasLooked = true;
            _cached = Look(alarm);
            return _cached;
        }

        /// <summary>Looks right now, ignoring the throttle, and refreshes the cache.</summary>
        public Transform LookNow(AlarmState alarm)
        {
            _cached = CanLook(alarm) ? Look(alarm) : null;
            _hasLooked = true;
            _untilNextLook = 1f / LooksPerSecond;
            return _cached;
        }

        private bool CanLook(AlarmState alarm)
        {
            if (_owner.IsIncapacitated)
                return false;
            return !(alarm == AlarmState.Calm && Time.time < s_arrivalGraceEndsAt);
        }

        private Transform Look(AlarmState alarm)
        {
            Transform guard = _owner.transform;
            Vector3 eye = guard.position + Vector3.up * _owner.EyeHeight;
            float range = GuardBrain.SightRange(_owner.SightRange, alarm);
            IReadOnlyList<Transform> intruders = _owner.Intruders;

            Transform best = null;
            float bestDistance = float.MaxValue;
            for (int i = 0; i < intruders.Count; i++)
            {
                Transform intruder = intruders[i];
                if (intruder == null)
                    continue;

                Vector3 target = intruder.position + Vector3.up * 1.0f;
                bool clear = !Physics.Linecast(eye, target, _owner.GeometryLayers);
                if (!GuardBrain.CanSee(eye, guard.forward, target, range, _owner.FieldOfView, clear))
                    continue;

                float distance = Vector3.Distance(eye, target);
                if (distance < bestDistance)
                {
                    bestDistance = distance;
                    best = intruder;
                }
            }
            return best;
        }
    }
}
